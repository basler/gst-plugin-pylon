# Common feature options
option('examples', type : 'feature', value : 'auto', yield : true)
option('tests', type : 'feature', value : 'auto', yield : true)
option('prototypes', type : 'feature', value : 'auto', yield : true)
option('gobject-cast-checks', type : 'feature', value : 'auto', yield : true,
       description: 'Enable run-time GObject cast checks (auto = enabled for development, disabled for stable releases)')
option('glib-asserts', type : 'feature', value : 'enabled', yield : true,
       description: 'Enable GLib assertion (auto = enabled for development, disabled for stable releases)')
option('glib-checks', type : 'feature', value : 'enabled', yield : true,
       description: 'Enable GLib checks such as API guards (auto = enabled for development, disabled for stable releases)')
option('python-bindings', type : 'feature', value : 'disabled', yield : true,
       description: 'Enable meta Python bindings')
option('pylon-root', type : 'string', value : '',
       description: 'Path to the Basler pylon SDK installation (falls back to PYLON_ROOT env var)')
option('pylon-rpath', type : 'string', value : '',
       description: 'Runtime library search path for installed Linux binaries; use "none" to omit (default: derived from pylon-root with cross sys_root stripped)')
option('nvmm', type : 'feature', value : 'auto',
       description: 'NVMM buffer support (DeepStream nvbufsurface + CUDA)')
option('nvds-incdir', type : 'string', value : '',
       description: 'Optional custom include directory containing nvbufsurface.h')
option('nvds-libdir', type : 'string', value : '',
       description: 'Optional custom library directory containing libnvbufsurface.so')
